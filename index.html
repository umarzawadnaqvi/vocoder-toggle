const videoElement = document.getElementById('input_video'); const statusDiv = document.getElementById('status'); const startBtn = document.getElementById('startBtn'); const debugLog = document.getElementById('debug-log'); function log(msg) { debugLog.innerText += "\n> " + msg; } let audioCtx, micSource, robotSynth, modulatorNode, outputGain; startBtn.onclick = async () => { log("Initializing Robot Voice..."); try { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); // 1. Get Microphone const stream = await navigator.mediaDevices.getUserMedia({ audio: true, video: true }); micSource = audioCtx.createMediaStreamSource(stream); videoElement.srcObject = stream; // 2. Create the "Robot Buzz" (The Carrier) robotSynth = audioCtx.createOscillator(); robotSynth.type = 'sawtooth'; robotSynth.frequency.setValueAtTime(150, audioCtx.currentTime); // Low robotic hum robotSynth.start(); // 3. The "Vocoder" Logic (Ring Modulation) // We use a GainNode where the mic controls the volume of the synth modulatorNode = audioCtx.createGain(); modulatorNode.gain.value = 0; // This connects your voice to the "Volume Knob" of the Buzz micSource.connect(modulatorNode.gain); robotSynth.connect(modulatorNode); // 4. Master Output (The Hand Toggle) outputGain = audioCtx.createGain(); outputGain.gain.value = 0; modulatorNode.connect(outputGain); outputGain.connect(audioCtx.destination); log("Hardware Linked. Loading AI..."); // 5. MediaPipe Hands const hands = new Hands({ locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/hands/${file}` }); hands.setOptions({ maxNumHands: 1, modelComplexity: 1 }); hands.onResults((results) => { if (results.multiHandLandmarks && results.multiHandLandmarks.length > 0) { const hand = results.multiHandLandmarks[0]; const isOpen = hand[8].y < hand[6].y; if (isOpen) { outputGain.gain.setTargetAtTime(1, audioCtx.currentTime, 0.05); statusDiv.innerText = "VOCODER: ACTIVE"; statusDiv.style.color = "#00ffcc"; } else { outputGain.gain.setTargetAtTime(0, audioCtx.currentTime, 0.05); statusDiv.innerText = "VOCODER: MUTED"; statusDiv.style.color = "#ff4444"; } } }); const camera = new Camera(videoElement, { onFrame: async () => { await hands.send({image: videoElement}); }, width: 640, height: 480 }); await camera.start(); log("System Online."); startBtn.style.display = 'none'; } catch (err) { log("ERROR: " + err.message); } };
